 ship(sh){const ctx=this.ctx,C=this.cell,cells=sh.cells;
  if(!cells||!cells.length)return;
  const hits=new Set((sh.hits||[]).map(p=>K(p[0],p[1])));
  const rem=cells.filter(p=>!hits.has(K(p[0],p[1])));
  let dead=hits.size>0&&rem.length===0;
  if(!dead&&hits.size>0&&rem.length&&this.shieldCells)dead=rem.every(p=>this.shieldCells.has(K(p[0],p[1])));
  if(sh.sunk)dead=true;
  const horiz=cells.length>1&&cells[0][0]===cells[1][0];
  const len=cells.length,L=len*C,W=C,ins=C*.12;
  const q0=this.xy(cells[0][0],cells[0][1]);
  const cx=q0[0]+(horiz?L:C)/2,cy=q0[1]+(horiz?C:L)/2,ph=cells[0][0]*2.3+cells[0][1]*1.7;
  ctx.save();
  if(dead){ctx.globalAlpha=.72;ctx.translate(cx,cy);ctx.rotate(.07);ctx.translate(-cx,-cy);}
  else if(CFG.wave>0&&CFG.fx){ctx.translate(cx,cy+Math.sin(this._t*.045+ph)*C*.02);
   ctx.rotate(Math.sin(this._t*.032+ph)*.013);ctx.translate(-cx,-cy);}
  ctx.translate(cx,cy);ctx.rotate(horiz?0:Math.PI/2);
  // локальные координаты: x — вдоль корабля (корма слева, нос справа), y — поперёк
  const hw=W-ins*2,y0=-hw/2,y1=hw/2,nose=Math.min(hw*1.05,C*.5);
  const hull=i2=>{const xa=-L/2+i2,xb=L/2-i2*1.4,ya=y0+i2*.85,yb=y1-i2*.85,nb=Math.max(3,nose-i2*1.6),r=(yb-ya)*.3;
   ctx.beginPath();ctx.moveTo(xa+r,ya);ctx.lineTo(xb-nb,ya);
   ctx.quadraticCurveTo(xb-nb*.15,ya+(yb-ya)*.1,xb,(ya+yb)/2);
   ctx.quadraticCurveTo(xb-nb*.15,yb-(yb-ya)*.1,xb-nb,yb);
   ctx.lineTo(xa+r,yb);ctx.quadraticCurveTo(xa,yb,xa,yb-r);
   ctx.lineTo(xa,ya+r);ctx.quadraticCurveTo(xa,ya,xa+r,ya);ctx.closePath();};
  const fx=f=>-L/2+L*f,t=this._t,foam=.16+.1*Math.sin(t*.05+ph);
  // пена вокруг корпуса
  ctx.strokeStyle='rgba(255,255,255,'+foam.toFixed(3)+')';ctx.lineWidth=Math.max(1.4,C*.05);
  ctx.beginPath();ctx.ellipse(0,0,L*.55,(W+ins*2)*.62,0,0,6.2832);ctx.stroke();
  // тень
  ctx.save();ctx.translate(2,3);ctx.fillStyle='rgba(8,40,56,.22)';hull(0);ctx.fill();ctx.restore();
  // корпус: скруглённая корма, прямые борта, острый нос
  const g=ctx.createLinearGradient(-L/2,0,L/2,0);
  g.addColorStop(0,dead?'#5a2030':'#25597a');g.addColorStop(.5,dead?'#8b2233':BP.hull);g.addColorStop(1,dead?'#6b3a44':'#3d7f9f');
  ctx.fillStyle=g;hull(0);ctx.fill();
  ctx.strokeStyle=dead?'rgba(255,180,190,.5)':'rgba(233,246,251,.75)';ctx.lineWidth=Math.max(1.2,C*.035);ctx.stroke();
  // палуба
  ctx.fillStyle=dead?'rgba(255,220,225,.25)':BP.deck;hull(C*.09);ctx.fill();
  ctx.strokeStyle=dead?'rgba(90,50,60,.4)':'rgba(11,34,48,.22)';ctx.lineWidth=Math.max(1,C*.026);hull(C*.09);ctx.stroke();
  if(!dead){ctx.strokeStyle='rgba(11,60,80,.14)';ctx.lineWidth=1;
   for(let i=1;i<=3;i++){const px=-L/2+L*i/4;
    ctx.beginPath();ctx.moveTo(px,y0+C*.14);ctx.lineTo(px,y1-C*.14);ctx.stroke();}}
  if(!dead){
   // надстройка с мостиком
   const b0=fx(.40),b1=fx(len===4?.64:(len===3?.62:.58)),bh=hw*.58;
   ctx.fillStyle=BP.trim;ctx.strokeStyle='rgba(140,90,10,.55)';ctx.lineWidth=Math.max(1,C*.028);
   this.rr(b0,-bh/2,b1-b0,bh,bh*.3);ctx.fill();ctx.stroke();
   ctx.fillStyle='#0d3a52';this.rr(b1-(b1-b0)*.26,-bh*.24,(b1-b0)*.16,bh*.48,(b1-b0)*.05);ctx.fill();
   // мачта с реем
   if(len>=3){const mx=fx(.32);
    ctx.strokeStyle='#8a5a2b';ctx.lineWidth=Math.max(1.2,C*.03);ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(mx,-hw*.36);ctx.lineTo(mx,hw*.36);ctx.stroke();
    ctx.fillStyle='#8a5a2b';ctx.beginPath();ctx.arc(mx,0,C*.045,0,6.2832);ctx.fill();}
   // трубы (у флагмана две, у крейсера одна)
   const fun=len===4?[.26,.16]:(len===3?[.22]:[]);
   fun.forEach((f,i)=>{const px=fx(f);
    ctx.fillStyle='#d8e6ee';ctx.strokeStyle='rgba(11,34,48,.5)';ctx.lineWidth=1;
    ctx.beginPath();ctx.arc(px,0,hw*.17,0,6.2832);ctx.fill();ctx.stroke();
    ctx.fillStyle='#13303f';ctx.beginPath();ctx.arc(px,0,hw*.10,0,6.2832);ctx.fill();
    ctx.strokeStyle=i?'#ff5d73':'#38e1ff';ctx.lineWidth=1.4;
    ctx.beginPath();ctx.arc(px,0,hw*.135,0,6.2832);ctx.stroke();});
   // башни с орудиями, стволами к носу
   const tur=len===4?[.78,.04]:(len===3?[.74]:[]);
   tur.forEach(f=>{const px=fx(f),tr=hw*.23;
    ctx.strokeStyle='#0d3a52';ctx.lineWidth=Math.max(1.6,C*.042);ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(px+tr*.6,-hw*.08);ctx.lineTo(px+tr+C*.18,-hw*.1);
    ctx.moveTo(px+tr*.6,hw*.08);ctx.lineTo(px+tr+C*.18,hw*.1);ctx.stroke();
    ctx.fillStyle='#33586e';ctx.strokeStyle='rgba(233,246,251,.6)';ctx.lineWidth=1;
    this.rr(px-tr,-tr,tr*2,tr*2,tr*.55);ctx.fill();ctx.stroke();
    ctx.fillStyle='rgba(233,246,251,.35)';ctx.beginPath();ctx.arc(px-tr*.2,-tr*.25,tr*.26,0,6.2832);ctx.fill();});
   // кормовой флаг
   {const px=fx(.06),fl=Math.sin(t*.05+ph)*C*.014;
    ctx.strokeStyle='#8a5a2b';ctx.lineWidth=Math.max(1,C*.026);
    ctx.beginPath();ctx.moveTo(px,0);ctx.lineTo(px,-C*.05);ctx.stroke();
    ctx.fillStyle='#ff5d73';ctx.beginPath();ctx.moveTo(px,-C*.05);
    ctx.lineTo(px-C*.16,-C*.015+fl);ctx.lineTo(px,C*.015);ctx.closePath();ctx.fill();}
   // носовые буруны
   ctx.strokeStyle='rgba(255,255,255,'+(foam*.9).toFixed(3)+')';ctx.lineWidth=Math.max(1,C*.032);ctx.lineCap='round';
   ctx.beginPath();ctx.moveTo(L/2-nose*.4,-hw*.5);ctx.quadraticCurveTo(L/2+C*.06,-hw*.2,L/2+C*.03,0);
   ctx.moveTo(L/2-nose*.4,hw*.5);ctx.quadraticCurveTo(L/2+C*.06,hw*.2,L/2+C*.03,0);ctx.stroke();}
  else{ // обломок: сломанная мачта и дым из пробоины
   const i0=cells.findIndex(p=>hits.has(K(p[0],p[1]))),hx0=i0>=0?-L/2+C*(i0+.5):0;
   ctx.strokeStyle='#5f6f7a';ctx.lineWidth=Math.max(1.6,C*.05);ctx.lineCap='round';
   ctx.beginPath();ctx.moveTo(fx(.3),0);ctx.lineTo(fx(.3)-C*.26,-C*.3);ctx.stroke();
   ctx.fillStyle='#7a8894';ctx.beginPath();ctx.moveTo(fx(.3)-C*.26,-C*.3);
   ctx.lineTo(fx(.3)-C*.12,-C*.2);ctx.lineTo(fx(.3)-C*.18,-C*.12);ctx.closePath();ctx.fill();
   if(CFG.fx){for(let i=0;i<3;i++){const off=((t*.55+i*27)%46);
    const sx2=hx0+Math.sin(t*.02+i*2.1)*C*.14,sy2=-C*.15-off*.5;
    ctx.globalAlpha=(1-off/46)*.4;ctx.fillStyle='#c9d8e0';
    ctx.beginPath();ctx.arc(sx2,sy2,(2.2+i*.8)*(C/52),0,6.2832);ctx.fill();}ctx.globalAlpha=.72;}}
  // пробоины и иллюминаторы по клеткам
  cells.forEach((p,i)=>{const xi=-L/2+C*(i+.5);
   if(hits.has(K(p[0],p[1]))){ctx.fillStyle='rgba(13,32,44,.85)';ctx.beginPath();ctx.arc(xi,0,C*.15,0,6.2832);ctx.fill();
    ctx.strokeStyle='rgba(255,140,70,.6)';ctx.lineWidth=Math.max(1,C*.03);
    ctx.beginPath();ctx.arc(xi,0,C*.15,0,6.2832);ctx.stroke();
    ctx.strokeStyle='rgba(10,25,36,.8)';ctx.lineWidth=Math.max(1,C*.026);
    ctx.beginPath();ctx.moveTo(xi-C*.1,-C*.08);ctx.lineTo(xi-C*.02,C*.05);
    ctx.moveTo(xi+C*.09,-C*.06);ctx.lineTo(xi+C*.01,C*.07);ctx.stroke();}
   else if(!dead){ctx.fillStyle='#0d3a52';
    ctx.beginPath();ctx.arc(xi,-hw*.26,C*.04,0,6.2832);ctx.fill();
    ctx.beginPath();ctx.arc(xi,hw*.26,C*.04,0,6.2832);ctx.fill();}});
  ctx.restore();}
 shieldDome(r,c){const ctx=this.ctx,C=this.cell,q=this.center(r,c),x=q[0],y=q[1];
  const pulse=.5+.5*Math.sin(this._t*.06);
  ctx.save();
  const g=ctx.createRadialGradient(x,y,C*.08,x,y,C*.42);
  g.addColorStop(0,'rgba(255,194,71,'+(.05+pulse*.05).toFixed(3)+')');
  g.addColorStop(.72,'rgba(255,194,71,'+(.15+pulse*.07).toFixed(3)+')');
  g.addColorStop(1,'rgba(255,194,71,'+(.3+pulse*.1).toFixed(3)+')');
  ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,C*.42,0,6.2832);ctx.fill();
  ctx.strokeStyle='rgba(255,194,71,'+(.55+pulse*.3).toFixed(3)+')';ctx.lineWidth=2;
  ctx.beginPath();ctx.arc(x,y,C*.42,0,6.2832);ctx.stroke();
  ctx.strokeStyle='rgba(255,224,130,.95)';ctx.lineWidth=Math.max(1.5,C*.04);ctx.lineJoin='round';
  ctx.beginPath();ctx.moveTo(x,y-C*.16);ctx.lineTo(x+C*.12,y-C*.11);ctx.lineTo(x+C*.12,y+C*.01);
  ctx.quadraticCurveTo(x,y+C*.17,x-C*.12,y+C*.01);ctx.lineTo(x-C*.12,y-C*.11);ctx.closePath();ctx.stroke();
  ctx.restore();}
