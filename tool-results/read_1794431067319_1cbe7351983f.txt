    13→/* ---------- Зернируемый генератор случайности ---------- */
    14→
    15→export function makeRng(seed: number): () => number {
    16→  let s = seed % 2147483647;
    17→  if (s <= 0) s += 2147483646;
    18→  return () => {
    19→    s = (s * 16807) % 2147483647;
    20→    return s / 2147483647;
    21→  };
    22→}
    23→
    24→export function seedFromText(text: string): number {
    25→  let h = 7;
    26→  for (let i = 0; i < text.length; i++) h = (h * 31 + text.charCodeAt(i)) % 2147483647;
    27→  return h;
    28→}

    32→export function fillRoundRect(ctx: Ctx2D, x: number, y: number, w: number, h: number, r: number): void {
    33→  const rr = Math.min(r, w / 2, h / 2);
    34→  ctx.beginPath();
    35→  ctx.moveTo(x + rr, y);
    36→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    37→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    38→  ctx.arcTo(x, y + h, x, y, rr);
    39→  ctx.arcTo(x, y, x + w, y, rr);
    40→  ctx.closePath();
    41→}
    42→
    43→export function strokeRoundRect(ctx: Ctx2D, x: number, y: number, w: number, h: number, r: number): void {
    44→  fillRoundRect(ctx, x, y, w, h, r);
    45→  ctx.stroke();
    46→}

    49→  ctx.beginPath();

    51→  ctx.closePath();

    60→/* Заполнить текущую фигуру цветом и обвести контуром */
    61→export function paint(ctx: Ctx2D, fill: string | null, stroke?: string, lw?: number): void {
    62→  if (fill) {
    63→    ctx.fillStyle = fill;
    64→    ctx.fill();
    65→  }
    66→  if (stroke) {
    67→    ctx.strokeStyle = stroke;
    68→    ctx.lineWidth = lw ?? 3;
    69→    ctx.lineJoin = 'round';
    70→    ctx.lineCap = 'round';
    71→    ctx.stroke();
    72→  }
    73→}

    94→export function drawSpaceBg(ctx: Ctx2D, s: number, rng: () => number): void {
    95→  const g = ctx.createRadialGradient(s * 0.5, s * 0.42, s * 0.1, s * 0.5, s * 0.5, s * 0.75);
    96→  g.addColorStop(0, '#31427e');
    97→  g.addColorStop(1, '#171c3f');
    98→  ctx.fillStyle = g;
    99→  ctx.fillRect(0, 0, s, s);
   100→
   101→  const neb = ctx.createRadialGradient(s * 0.72, s * 0.24, 0, s * 0.72, s * 0.24, s * 0.34);
   102→  neb.addColorStop(0, 'rgba(150,110,220,0.35)');
   103→  neb.addColorStop(1, 'rgba(150,110,220,0)');
   104→  ctx.fillStyle = neb;
   105→  ctx.fillRect(0, 0, s, s);
   106→  const neb2 = ctx.createRadialGradient(s * 0.2, s * 0.7, 0, s * 0.2, s * 0.7, s * 0.3);
   107→  neb2.addColorStop(0, 'rgba(80,160,220,0.25)');
   108→  neb2.addColorStop(1, 'rgba(80,160,220,0)');
   109→  ctx.fillStyle = neb2;
   110→  ctx.fillRect(0, 0, s, s);
   111→
   112→  for (let i = 0; i < 70; i++) {
   113→    const x = rng() * s;
   114→    const y = rng() * s;
   115→    const r = (0.004 + rng() * 0.01) * s;
   116→    ctx.fillStyle = rng() > 0.85 ? 'rgba(255,230,150,0.95)' : 'rgba(255,255,255,0.85)';
   117→    circle(ctx, x, y, r);
   118→    ctx.fill();
   119→  }
   120→  for (let i = 0; i < 5; i++) {
   121→    const x = rng() * s;
   122→    const y = rng() * s * 0.8;
   123→    const r = s * 0.016;
   124→    ctx.strokeStyle = 'rgba(255,255,255,0.9)';
   125→    ctx.lineWidth = s * 0.004;
   126→    ctx.lineCap = 'round';
   127→    ctx.beginPath();
   128→    ctx.moveTo(x - r, y); ctx.lineTo(x + r, y);
   129→    ctx.moveTo(x, y - r); ctx.lineTo(x, y + r);
   130→    ctx.stroke();
   131→  }
   132→}

   136→export function drawSun(ctx: Ctx2D, s: number, pal: PuzzlePalette, x = 0.82, y = 0.15, r = 0.075): void {
   137→  const cx = s * x;
   138→  const cy = s * y;
   139→  const cr = s * r;
   140→  ctx.save();
   141→  ctx.strokeStyle = 'rgba(255,200,60,0.85)';
   142→  ctx.lineWidth = s * 0.011;
   143→  ctx.lineCap = 'round';
   144→  for (let i = 0; i < 12; i++) {
   145→    const a = (i * Math.PI) / 6;
   146→    ctx.beginPath();
   147→    ctx.moveTo(cx + Math.cos(a) * cr * 1.35, cy + Math.sin(a) * cr * 1.35);
   148→    ctx.lineTo(cx + Math.cos(a) * cr * 1.8, cy + Math.sin(a) * cr * 1.8);
   149→    ctx.stroke();
   150→  }
   151→  const g = ctx.createRadialGradient(cx - cr * 0.3, cy - cr * 0.3, cr * 0.2, cx, cy, cr);
   152→  g.addColorStop(0, '#fff3bf');
   153→  g.addColorStop(1, pal.sun);
   154→  ctx.fillStyle = g;
   155→  circle(ctx, cx, cy, cr);
   156→  ctx.fill();
   157→  ctx.strokeStyle = 'rgba(230,145,20,0.5)';
   158→  ctx.lineWidth = s * 0.004;
   159→  ctx.stroke();
   160→  ctx.restore();
   161→}

   190→export function drawHillBand(
   191→  ctx: Ctx2D, s: number, base: number, amp: number, color: string,
   192→  rng: () => number, waves = 3,
   193→): void {
   194→  ctx.fillStyle = color;
   195→  ctx.beginPath();
   196→  ctx.moveTo(0, s);
   197→  ctx.lineTo(0, s * base);
   198→  for (let w = 0; w < waves; w++) {
   199→    const x0 = (s * w) / waves;
   200→    const x1 = (s * (w + 1)) / waves;
   201→    const peak = s * (base - amp - rng() * amp * 0.6);
   202→    ctx.quadraticCurveTo((x0 + x1) / 2, peak, x1, s * base);
   203→  }
   204→  ctx.lineTo(s, s);
   205→  ctx.closePath();
   206→  ctx.fill();
   207→}

   233→  ctx.fill();
   234→  ctx.strokeStyle = outline;
   235→  ctx.lineWidth = s * 0.005;
   236→  ctx.lineJoin = 'round';
   237→  ctx.stroke();
   238→  ctx.beginPath();

   244→  ctx.quadraticCurveTo(x + w * 0.12, base - h * 0.68, x + w * 0.02, base - h * 0.6);
   245→  ctx.quadraticCurveTo(x - w * 0.12, base - h * 0.7, x - w * 0.3, base - h * 0.62);

   319→      topX + Math.cos(a) * h * 0.42, topY + Math.sin(a) * h * 0.34,

   323→      topX, topY,
   324→    );
   325→    ctx.fill();
   326→    ctx.stroke();
   327→  }

   333→export function drawCitySkyline(ctx: Ctx2D, s: number, base: number, rng: () => number, color: string): void {
   334→  let x = 0;
   335→  while (x < s) {
   336→    const w = s * (0.07 + rng() * 0.09);
   337→    const h = s * (0.08 + rng() * 0.16);
   338→    ctx.fillStyle = color;
   339→    ctx.fillRect(x, base - h, w, h);
   340→    ctx.fillStyle = 'rgba(255,240,180,0.75)';
   341→    for (let wy = base - h + s * 0.015; wy < base - s * 0.015; wy += s * 0.035) {
   342→      for (let wx = x + s * 0.012; wx < x + w - s * 0.02; wx += s * 0.03) {
   343→        ctx.fillRect(wx, wy, s * 0.014, s * 0.018);
   344→      }
   345→    }
   346→    x += w + s * 0.015;
   347→  }
   348→}

   467→function paintCountryside(c: SceneContext, horizon: number): void {
   468→  const { ctx, s, rng, pal } = c;
   469→  drawSky(ctx, s, pal);
   470→  drawSun(ctx, s, pal);
   471→  drawCloud(ctx, s, s * 0.18, s * 0.14, s * 0.035);
   472→  drawCloud(ctx, s, s * 0.52, s * 0.2, s * 0.028, 0.8);
   473→  drawHillBand(ctx, s, horizon - 0.1, 0.05, pal.hillFar, rng, 2);
